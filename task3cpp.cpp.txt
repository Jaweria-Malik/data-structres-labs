#include<iostream>
using namespace std;
class carstack {
private:
	int arr[8];
	int top;
public:
	carstack() {
		top = -1;
	}
	bool isempty() {
		return top == -1;
	}
	bool isfull() {
		return top == 7;
	}
	void push(int car) {
		if (isfull()) {
			cout << "PARKING IS FULL" << endl;
			return;
		}
		arr[++top] = car;
	}
	int pop() {
		if (isempty()) {
			return -1;
		}
		return arr[top--];
	}
	int peek() {
	if (isempty()) {
		return -1;
	}
	return arr[top];
}

void display() {
	if (isempty()) {
		cout << "No cars parked\n";
		return;
	}
	for (int i = top; i >= 0; i--) {
		cout << arr[i] << " ";
	}
	cout << endl;
}

bool search(int car) {
	for (int i = 0; i <= top; i++) {
		if (arr[i] == car)
			return true;
	}
	return false;
}

int getTopIndex() {
	return top;
}
};
int main() {
	carstack parking, temp;
	int choice, car;

	do {
		cout << "\n1.Park Car";
		cout << "\n2.Remove Car";
		cout << "\n3.Display Cars";
		cout << "\n4.Search Car";
		cout << "\n5.Total Cars";
		cout << "\n6.Exit\n";

		cin >> choice;

		switch (choice) {
		case 1:
			cout << "Enter car number: ";
			cin >> car;
			parking.push(car);
			break;
		case 2:
			cout << "Enter car number to remove: ";
			cin >> car;

			if (!parking.search(car)) {
				cout << "Car not found\n";
				break;
			}

			while (parking.peek() != car) {
				temp.push(parking.pop());
			}

			parking.pop();

			while (!temp.isempty()) {
				parking.push(temp.pop());
			}

			cout << "Car removed\n";
			break;

		case 3:
			parking.display();
			break;
		case 4:
			cout << "Enter car number: ";
			cin >> car;
			if (parking.search(car))
				cout << "Car Found\n";
			else
				cout << "Car Not Found\n";
			break;

		case 5:
			cout << "Total Cars: " << parking.getTopIndex() + 1 << endl;
			break;

		}

	} while (choice != 6);

	return 0;
}
