#include <iostream>
using namespace std;

class CharStack {
private:
    char arr[100];
    int top;

public:
    CharStack() { 
        top = -1; }
    bool isEmpty() {
        return top == -1; }
    bool isFull() {
        return top == 99; }
    void push(char c) {
        if (!isFull()) arr[++top] = c; }
    char pop() {
        return isEmpty() ? '\0' : arr[top--]; }
    char peek() {
        return isEmpty() ? '\0' : arr[top]; }
    void clear() {
        top = -1; }
};

int main() {
    CharStack undoStack, redoStack;
    char text[100];
    int len = 0;
    int choice;
    char ch;

    do {
        cout << "\n1.Type Character";
        cout << "\n2.Delete Character";
        cout << "\n3.Undo";
        cout << "\n4.Redo";
        cout << "\n5.Show Text";
        cout << "\n6.Exit\n";

        cout << "Enter choice: ";
        cin >> choice;

        switch (choice) {
        case 1:
            cout << "Enter character: ";
            cin >> ch;
            text[len++] = ch;
            undoStack.push(ch);
            redoStack.clear();
            break;

        case 2:
            if (len > 0) {
                char removed = text[--len];
                redoStack.push(removed);
            }
            break;

        case 3:
            if (!undoStack.isEmpty()) {
                char last = undoStack.pop();
                len--;
                redoStack.push(last);
            }
            break;

        case 4:
            if (!redoStack.isEmpty()) {
                char r = redoStack.pop();
                text[len++] = r;
                undoStack.push(r);
            }
            break;

        case 5:
            cout << "Text: ";
            for (int i = 0; i < len; i++)
                cout << text[i];
            cout << endl;
            break;
        }

    } while (choice != 6);

    return 0;
}
